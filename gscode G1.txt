function doGet(e) {
  var page = e && e.parameter ? e.parameter.page : '';
  if (page === 'manifest') return serveWebManifest();
  if (page === 'sw') return serveServiceWorker();
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('جدول مرضى غسيل الكلى')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function serveWebManifest() {
  var url = ScriptApp.getService().getUrl();
  var iconSvg = '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 192 192"><rect width="192" height="192" rx="36" fill="#1565c0"/><path fill="#ffffff" d="M96 36c-22 28-44 48-44 72a44 44 0 1088 0c0-24-22-44-44-72z"/></svg>';
  var manifest = {
    name: 'جدول مرضى غسيل الكلى',
    short_name: 'جدول الغسيل',
    description: 'إدارة مواعيد مرضى غسيل الكلى',
    start_url: url,
    scope: url,
    display: 'standalone',
    background_color: '#eef2f7',
    theme_color: '#1565c0',
    orientation: 'any',
    dir: 'rtl',
    lang: 'ar',
    icons: [
      {
        src: 'data:image/svg+xml,' + encodeURIComponent(iconSvg),
        sizes: '192x192',
        type: 'image/svg+xml',
        purpose: 'any maskable'
      }
    ]
  };
  return ContentService.createTextOutput(JSON.stringify(manifest))

}

function serveServiceWorker() {
  var sw = [
    "self.addEventListener('install', function(e) { self.skipWaiting(); });",
    "self.addEventListener('activate', function(e) { e.waitUntil(self.clients.claim()); });"
  ].join('\n');
  return ContentService.createTextOutput(sw).setMimeType(ContentService.MimeType.JAVASCRIPT);
}

// الجدول الثابت: الصف 103 (عناوين) إلى 199 (آخر P+ شفت 4) — نفس هيكل المتغير (103-199 = 97 صف)
var STATIC_SCHEDULE_START = 102; // فهرس 0-based لبداية الصف 103
var STATIC_SCHEDULE_ROW_COUNT = 97;
var STATIC_SHEET_ROW_OFFSET = 102; // فرق صف الشيت: staticSheetRow = variableSheetRow + 102

function normalizeNameKey(name) {
  if (!name) return "";
  return name.toString().trim().replace(/\s+/g, " ").toLowerCase();
}

function namesMatchExact(cellName, patientName) {
  if (!cellName || !patientName) return false;
  return normalizeNameKey(cellName) === normalizeNameKey(patientName);
}

function findInfoHeaderRow(infoData) {
  for (var r = 0; r < Math.min(infoData.length, 30); r++) {
    var rowText = infoData[r].map(function(c) { return c ? c.toString() : ""; }).join("|");
    if (rowText.indexOf("اسم المريض") !== -1) return r;
  }
  return 0;
}

function buildPatientObject(infoData, i, headerRowIdx, headers) {
  var pObj = {};
  headers.forEach(function(header, index) {
    if (header) pObj[header.toString().trim()] = infoData[i][index];
  });
  if (!pObj["اسم المريض"] && infoData[i][12]) pObj["اسم المريض"] = infoData[i][12].toString().trim();
  pObj["رقم الطبلة"] = pObj["رقم الطبلة"] || (infoData[i][13] ? infoData[i][13].toString().trim() : "");
  pObj["الجنس"] = pObj["الجنس"] || (infoData[i][11] ? infoData[i][11].toString().trim() : "");
  pObj["العمر"] = pObj["العمر"] || (infoData[i][10] ? infoData[i][10].toString().trim() : "");
  pObj["المهنة"] = pObj["المهنة"] || (infoData[i][9] ? infoData[i][9].toString().trim() : "");
  pObj["عنوان السكن"] = pObj["عنوان السكن"] || (infoData[i][8] ? infoData[i][8].toString().trim() : "");
  pObj["الطبيب"] = pObj["الطبيب"] || (infoData[i][7] ? infoData[i][7].toString().trim() : "");
  pObj["الوزن"] = pObj["الوزن"] || (infoData[i][6] ? infoData[i][6].toString().trim() : "");
  pObj["الفلتر"] = pObj["الفلتر"] || pObj["Filter"] || (infoData[i][5] ? infoData[i][5].toString().trim() : "");
  pObj["رقم الموبايل 1"] = pObj["رقم الموبايل 1"] || (infoData[i][4] ? infoData[i][4].toString().trim() : "");
  pObj["رقم الموبايل 2"] = pObj["رقم الموبايل 2"] || (infoData[i][3] ? infoData[i][3].toString().trim() : "");
  pObj["Fistula"] = pObj["Fistula"] || (infoData[i][2] ? infoData[i][2].toString().trim() : "");
  pObj["ملاحظات"] = pObj["ملاحظات"] || (infoData[i][14] ? infoData[i][14].toString().trim() : "");
  return pObj;
}

function getUnifiedData() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("الجدول");
    if (!sheet) return { success: false, error: "لم يتم العثور على ورقة 'الجدول'" };
    var lastRow = sheet.getLastRow();

    var range = sheet.getRange(1, 1, Math.max(lastRow, 350), 13);
    var dataValues = range.getValues();
    var dataColors = range.getBackgrounds();
    var monthValue = dataValues[1][9] || "---";

    var generalNotes = [];
    var notesSheet = ss.getSheetByName("ملاحظات الكادر");
    if (notesSheet) {
      // قراءة العمودين I و J (نص الملاحظة وتوقيت انتهاء الملاحظة المحدد)
      var notesValues = notesSheet.getRange(2, 9, 15, 2).getValues();
      var now = new Date().getTime();
      var anyExpired = false;
      for (var ni = 0; ni < 15; ni++) {
        var noteText = notesValues[ni][0] ? notesValues[ni][0].toString().trim() : "";
        var expiry = notesValues[ni][1] ? new Date(notesValues[ni][1]).getTime() : null;
        
        if (noteText) {
          if (expiry && now > expiry) {
            // إذا انتهى تاريخ ووقت صلاحية الملاحظة المحدد، يمسحها السيرفر تلقائياً من الشيت
            notesSheet.getRange(ni + 2, 9).setValue("");
            notesSheet.getRange(ni + 2, 10).setValue("");
            anyExpired = true;
          } else {
            generalNotes.push({ 
              text: noteText, 
              row: ni + 2,
              expiry: notesValues[ni][1] ? notesValues[ni][1].toString() : ""
            });
          }
        }
      }
      if (anyExpired) {
        SpreadsheetApp.flush();
      }
    }

    var schedule = dataValues.slice(0, 98);
    var scheduleColors = dataColors.slice(0, 98);
    var staticSchedule = dataValues.slice(STATIC_SCHEDULE_START, STATIC_SCHEDULE_START + STATIC_SCHEDULE_ROW_COUNT);
    var staticColors = dataColors.slice(STATIC_SCHEDULE_START, STATIC_SCHEDULE_START + STATIC_SCHEDULE_ROW_COUNT);

    var infoSheet = ss.getSheetByName("معلومات المرضى") || ss.getSheetByName("جدول_1") || ss.getSheets()[1];
    var patientsDetailedInfo = [];
    if (infoSheet) {
      var infoData = infoSheet.getDataRange().getValues();
      var headerRowIdx = findInfoHeaderRow(infoData);
      var headers = infoData[headerRowIdx];
      for (var i = headerRowIdx + 1; i < infoData.length; i++) {
        var pObj = buildPatientObject(infoData, i, headerRowIdx, headers);
        if (pObj["اسم المريض"]) patientsDetailedInfo.push(pObj);
      }
    }

    var contactsMap = {};
    patientsDetailedInfo.forEach(function(p) {
      var fullName = p["اسم المريض"] ? p["اسم المريض"].toString().trim() : "";
      if (!fullName) return;
      contactsMap[fullName] = {
        notes: p["ملاحظات"] || "",
        phone1: p["رقم الموبايل 1"] || "",
        phone2: p["رقم الموبايل 2"] || "",
        gender: p["الجنس"] || "",
        filter: p["الفلتر"] || "",
        fistula: p["Fistula"] || ""
      };
    });

    var tabletByName = {};
    patientsDetailedInfo.forEach(function(p) {
      var fullName = p["اسم المريض"] ? p["اسم المريض"].toString().trim() : "";
      if (!fullName) return;
      var tablet = p["رقم الطبلة"];
      if (tablet !== undefined && tablet !== null && tablet !== "") {
        tabletByName[normalizeNameKey(fullName)] = tablet;
      }
    });

    var patientNames = Object.keys(contactsMap).filter(function(n) { return n; });
    patientNames.sort(function(a, b) { return a.localeCompare(b, "ar"); });

    var daysInArabic = ["الأحد", "الاثنين", "الثلاثاء", "الأربعاء", "الخميس", "الجمعة", "السبت"];
    var todayName = daysInArabic[new Date().getDay()];
    var dayMap = { "السبت": 1, "الأحد": 2, "الاثنين": 3, "الثلاثاء": 4, "الأربعاء": 5, "الخميس": 6, "الجمعة": 7 };

    // جلب مؤقتات جلسات المرضى السحابية المشتركة وتنظيف القديم تلقائياً بعد ساعة كاملة من الانتهاء
    var scriptProps = PropertiesService.getScriptProperties();
    var timersJson = scriptProps.getProperty("patient_timers") || "{}";
    var timers = JSON.parse(timersJson);
    var now = new Date().getTime();
    var cleanedTimers = {};
    
    for (var name in timers) {
      var expiry = new Date(timers[name]).getTime();
      // تمديد الاحتفاظ بالمؤقتات في السيرفر السحابي لتغطي ساعة كاملة (60 دقيقة) بعد الاكتمال
      if (expiry - now > -60 * 60 * 1000) {
        cleanedTimers[name] = timers[name];
      }
    }
    
    if (Object.keys(timers).length !== Object.keys(cleanedTimers).length) {
      scriptProps.setProperty("patient_timers", JSON.stringify(cleanedTimers));
    }

    return {
      success: true,
      month: monthValue,
      genNotes: generalNotes,
      schedule: schedule,
      colors: scheduleColors,
      staticSchedule: staticSchedule,
      staticColors: staticColors,
      staticOffset: STATIC_SHEET_ROW_OFFSET,
      contacts: contactsMap,
      today: dayMap[todayName],
      patientsDetailedInfo: patientsDetailedInfo,
      tabletByName: tabletByName,
      patientNames: patientNames,
      fieldSuggestions: getFieldSuggestionsFromPatients(patientsDetailedInfo),
      theme: getUserTheme(),
      timers: cleanedTimers
    };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function getFieldSuggestionsFromPatients(patientsDetailedInfo) {
  var fields = {
    gender: {},
    doctor: {},
    address: {},
    job: {},
    filter: {},
    fistula: {}
  };
  (patientsDetailedInfo || []).forEach(function(p) {
    if (p["الجنس"]) fields.gender[p["الجنس"].toString().trim()] = true;
    if (p["الطبيب"]) fields.doctor[p["الطبيب"].toString().trim()] = true;
    if (p["عنوان السكن"]) fields.address[p["عنوان السكن"].toString().trim()] = true;
    if (p["المهنة"]) fields.job[p["المهنة"].toString().trim()] = true;
    if (p["الفلتر"]) fields.filter[p["الفلتر"].toString().trim()] = true;
    if (p["Fistula"]) fields.fistula[p["Fistula"].toString().trim()] = true;
  });
  var result = {};
  Object.keys(fields).forEach(function(k) {
    result[k] = Object.keys(fields[k]).sort(function(a, b) { return a.localeCompare(b, "ar"); });
  });
  return result;
}

function isNoteReadStatus(value) {
  var v = value ? value.toString().trim() : "";
  return v === "مقروء" || v === "1" || v.toLowerCase() === "read" || v.toLowerCase() === "true";
}

function verifyAdminKey(userPassword) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("ملاحظات الكادر");
    if (!sheet) return { success: false, error: "ورقة الملاحظات غير موجودة" };

      return { success: true, authenticated: true };
    }
    return { success: true, authenticated: false };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function getStaffNotesCount() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("ملاحظات الكادر");
    if (!sheet) return 0;
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) return 0;
    var numRows = lastRow - 1;
    var data = sheet.getRange(2, 2, numRows, 4).getValues();
    var count = 0;
    for (var i = 0; i < data.length; i++) {
      var text = data[i][0] ? data[i][0].toString().trim() : "";
      var readVal = data[i][3];
      if (text && !isNoteReadStatus(readVal)) count++;
    }
    return count;
  } catch (e) {
    return 0;
  }
}

function getStaffNotes() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("ملاحظات الكادر");
    if (!sheet) return { success: false, error: "لم يتم العثور على ورقة 'ملاحظات الكادر'" };
    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) return { success: true, notes: [] };
    var numRows = lastRow - 1;
    var data = sheet.getRange(2, 1, numRows, 5).getValues();
    var allNotes = [];
    for (var i = data.length - 1; i >= 0; i--) {
      var day = data[i][0] ? data[i][0].toString().trim() : "";
      var text = data[i][1] ? data[i][1].toString().trim() : "";
      var rawDate = data[i][2];
      var sender = data[i][3] ? data[i][3].toString().trim() : "مجهول";
      var readVal = data[i][4];
      var isRead = isNoteReadStatus(readVal);
      if (text) {
        var timeStr = "";
        if (rawDate instanceof Date) {
          timeStr = Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy/MM/dd hh:mm a");
        } else if (rawDate) {
          timeStr = rawDate.toString();
        }
        allNotes.push({
          row: i + 2,
          day: day || "غير محدد",
          text: text,
          time: timeStr,
          sender: sender,
          read: isRead
        });
      }
    }
    return { success: true, notes: allNotes };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function saveScheduleChanges(adminPassword, changes, isStaticSchedule) {
  try {
    var authResult = verifyAdminKey(adminPassword);
    if (!authResult.success) return authResult;
    if (!authResult.authenticated) {
      return { success: false, error: "الرمز السري غير صحيح" };
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("الجدول");
    if (!sheet) return { success: false, error: "لم يتم العثور على ورقة 'الجدول'" };
    if (!changes || !changes.length) {
      return { success: true, message: "لا توجد تغييرات للحفظ" };
    }
    var rowShift = isStaticSchedule ? STATIC_SHEET_ROW_OFFSET : 0;
    changes.forEach(function(ch) {
      var r = parseInt(ch.row, 10) + 1 + rowShift;
      var c = parseInt(ch.col, 10) + 1;
      if (r < 1 || c < 1) return;
      var cellRange = sheet.getRange(r, c);
      if (Object.prototype.hasOwnProperty.call(ch, "value")) {
        cellRange.setValue(ch.value || "");
      }
      if (Object.prototype.hasOwnProperty.call(ch, "color")) {
        cellRange.setBackground(ch.color || "#ffffff");
      }
    });
    SpreadsheetApp.flush();
    return { success: true };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function copyStaticToVariableSchedule(copyPassword) {
  try {

    if (!authResult.authenticated) {

    return { success: false, error: e.toString() };
  }

function saveGeneralNotes(adminPassword, notesArray) {
  try {
    var authResult = verifyAdminKey(adminPassword);

      return { success: false, error: "الرمز السري غير صحيح" };

function findFirstEmptyInfoRow(infoSheet, headerRowIdx) {
  var infoData = infoSheet.getDataRange().getValues();
  for (var i = headerRowIdx + 1; i < infoData.length; i++) {
    var nameCell = infoData[i][12] ? infoData[i][12].toString().trim() : "";
    if (!nameCell) return i + 1;
  }
  return infoSheet.getLastRow() + 1;
}

function addOrUpdatePatientRow(adminPassword, pData, isNew) {
  try {
    var authResult = verifyAdminKey(adminPassword);
    if (!authResult.success || !authResult.authenticated) {
      return { success: false, error: "الرمز السري غير صحيح" };
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var infoSheet = ss.getSheetByName("معلومات المرضى") || ss.getSheetByName("جدول_1") || ss.getSheets()[1];
    if (!infoSheet) return { success: false, error: "لم يتم العثور على ورقة معلومات المرضى" };

    var infoData = infoSheet.getDataRange().getValues();
    var headerRowIdx = findInfoHeaderRow(infoData);
    var targetRow = -1;

    if (!isNew && pData.originalName) {
      for (var i = headerRowIdx + 1; i < infoData.length; i++) {
        var currentName = infoData[i][12] ? infoData[i][12].toString().trim() : "";
        if (namesMatchExact(currentName, pData.originalName)) {
          targetRow = i + 1;
          break;
        }
      }
    }

    if (targetRow === -1) {
      targetRow = findFirstEmptyInfoRow(infoSheet, headerRowIdx);
      if (targetRow > infoSheet.getLastRow()) {
        infoSheet.getRange(targetRow, 1).setValue(targetRow - headerRowIdx - 1);
      }
    }

    infoSheet.getRange(targetRow, 13).setValue(pData.fullName);
    infoSheet.getRange(targetRow, 14).setValue(pData.tabletNum);
    var genderVal = pData.gender ? pData.gender.toString().trim() : "";
    if (genderVal && genderVal !== "ذكر" && genderVal !== "أنثى") {
      return { success: false, error: "الجنس يجب أن يكون ذكر أو أنثى فقط" };
    }
    infoSheet.getRange(targetRow, 12).setValue(genderVal);
    infoSheet.getRange(targetRow, 11).setValue(pData.age);

function deletePatientFromServer(adminPassword, fullName) {
  try {
    var authResult = verifyAdminKey(adminPassword);
    if (!authResult.success || !authResult.authenticated) {
      return { success: false, error: "الرمز السري غير صحيح" };
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    var infoSheet = ss.getSheetByName("معلومات المرضى") || ss.getSheetByName("جدول_1") || ss.getSheets()[1];
    if (infoSheet) {
      var infoData = infoSheet.getDataRange().getValues();
      for (var i = infoData.length - 1; i >= 0; i--) {
        var currentName = infoData[i][12] ? infoData[i][12].toString().trim() : "";
        if (namesMatchExact(currentName, fullName)) {
          infoSheet.deleteRow(i + 1);
        }
      }
    }

    var sheet = ss.getSheetByName("الجدول");
    if (sheet) {
      var scheduleRange = sheet.getRange(2, 2, 96, 7);
      var scheduleValues = scheduleRange.getValues();
      var scheduleColors = scheduleRange.getBackgrounds();
      for (var r = 0; r < scheduleValues.length; r++) {
        for (var c = 0; c < 7; c++) {
          if (scheduleValues[r][c] && namesMatchExact(scheduleValues[r][c].toString().trim(), fullName)) {
            sheet.getRange(r + 2, c + 2).setValue("");
            sheet.getRange(r + 2, c + 2).setBackground("#ffffff");
          }
        }
      }
    }

    SpreadsheetApp.flush();
    return { success: true };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function updatePatientNote(adminPassword, fullName, noteText) {
  try {
    var authResult = verifyAdminKey(adminPassword);
    if (!authResult.success || !authResult.authenticated) {
      return { success: false, error: "الرمز السري غير صحيح" };
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var infoSheet = ss.getSheetByName("معلومات المرضى") || ss.getSheetByName("جدول_1") || ss.getSheets()[1];
    if (!infoSheet) return { success: false, error: "لم يتم العثور على ورقة معلومات المرضى" };

    return { success: true };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

// دالة حفظ وتحديث مؤقت جلسة المريض سحابياً في PropertiesService
function setPatientTimerServer(patientName, expiryTime) {
  try {
    var scriptProps = PropertiesService.getScriptProperties();
    var timersJson = scriptProps.getProperty("patient_timers") || "{}";
    var timers = JSON.parse(timersJson);
    
    if (expiryTime === "") {
      delete timers[patientName];
    } else {
      timers[patientName] = expiryTime;
    }
    
    scriptProps.setProperty("patient_timers", JSON.stringify(timers));
    return { success: true };
  } catch (e) {
    return { success: false, error: e.message };
  }
}

// دالة استقبال ومعالجة الطلبات القادمة بأمان من موقع جيت هاب الخارجي (CORS API Gateway)
function doPost(e) {
  try {
    var postData = JSON.parse(e.postData.contents);
    var functionName = postData.functionName;
    var args = postData.arguments || [];
    
    // استدعاء الدالة المطلوبة ديناميكياً وتمرير المعاملات المرفقة لها
    var result = this[functionName].apply(this, args);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      result: result
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      error: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
